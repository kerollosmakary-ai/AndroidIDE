<script>
const searchableData = [
  { title: "Navbar Component", desc: "Glassmorphic responsive navigation bar", icon: "fa-compass", link: "#home" },
  { title: "Hero Section", desc: "Modern hero section with action buttons", icon: "fa-rocket", link: "#home" },
  { title: "Services Cards", desc: "Grid list of services with icons", icon: "fa-cubes", link: "#services" },
  { title: "Stats Counter", desc: "Social proof statistics display", icon: "fa-chart-line", link: "#stats" },
  { title: "Portfolio Carousel", desc: "Image slider with captions", icon: "fa-images", link: "#portfolio" },
  { title: "FAQ Accordion", desc: "Collapsible questions and answers", icon: "fa-circle-question", link: "#faq" },
  { title: "Contact Form", desc: "Interactive glass form interface", icon: "fa-paper-plane", link: "#contact" },
  { title: "Footer Section", desc: "Footer links with social icons", icon: "fa-anchor", link: "#footer" }
];

const searchInput = document.getElementById('searchInput');
const searchResults = document.getElementById('searchResults');

searchInput.addEventListener('input', function(e) {
  const query = e.target.value.toLowerCase().trim();
  if (query.length === 0) {
    searchResults.classList.remove('active');
    searchResults.innerHTML = '';
    return;
  }
  const filteredData = searchableData.filter(item => 
    item.title.toLowerCase().includes(query) || 
    item.desc.toLowerCase().includes(query)
  );
  renderResults(filteredData);
});

function renderResults(results) {
  if (results.length === 0) {
    searchResults.innerHTML = '<div class="no-result"><i class="fa-solid fa-triangle-exclamation"></i> No matching components found.</div>';
  } else {
    searchResults.innerHTML = results.map(item => `
      <a href="${item.link}" class="result-item">
        <i class="fa-solid ${item.icon} result-icon"></i>
        <div class="result-info">
          <h4>${item.title}</h4>
          <p>${item.desc}</p>
        </div>
      </a>
    `).join('');
  }
  searchResults.classList.add('active');
}

document.addEventListener('click', function(e) {
  if (!e.target.closest('.search-container')) {
    searchResults.classList.remove('active');
  }
});
</script>
